package io.github.simplejdbcmapper.core;

import io.github.simplejdbcmapper.exception.MapperException;
import java.lang.reflect.Constructor;
import java.sql.ResultSet;
import java.sql.SQLException;
import java.sql.SQLFeatureNotSupportedException;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.core.convert.ConversionService;
import org.springframework.jdbc.core.RowMapper;
import org.springframework.jdbc.support.JdbcUtils;

/**
 * A row mapper for mapped objects.
 *
 * <p>Use the method {@link io.github.simplejdbcmapper.core.SimpleJdbcMapper#entityRowMapper} to get
 * an instance of EntityRowMapper.
 *
 * <p>EntityRowMapper <b>always</b> has to be used with sql constructed from either
 *
 * <pre>
 *    {@link io.github.simplejdbcmapper.core.SimpleJdbcMapper#select(Class<?> entityType)} or
 *    {@link io.github.simplejdbcmapper.core.SimpleJdbcMapper#select(Class<?> entityType, String tableAlias)}
 * </pre>
 *
 * EntityRowMapper expects the sql columns to be in a specific order since it uses position indexes
 * to retrieve the data from the query ResultSet. <b>Do not modify the sql from the select()
 * method</b>.
 *
 * <p>It will handle the cases where the column to property mappings do not follow the underscore to
 * camel case naming convention.
 *
 * <p>This is the recommended row mapper to use when writing custom queries for mapped objects. It
 * has optimizations which allows it to process data from the query ResultSet in a performant way.
 *
 * <p>Example query:
 *
 * <pre>
 * String sql = sjm.select(Product.class).clauses("FROM product WHERE name = ?");
 * {@code List<Product>} products = sjm.getJdbcTemplate().query(sql, sjm.entityRowMapper(Product.class), "someProductName");
 * </pre>
 *
 * @param <T> the entityType
 * @author Antony Joseph
 */
public final class EntityRowMapper<T> implements RowMapper<T> {
  private static final Logger logger = LoggerFactory.getLogger(EntityRowMapper.class);

  private final ConversionService conversionService;
  private final PropertyMapping[] propertyMappings;
  private final Constructor<T> mappedObjConstructor;
  private final int startIndex;
  private final int endIndex;

  @SuppressWarnings("unchecked")
  EntityRowMapper(TableMapping tableMapping, ConversionService conversionService, int offset) {
    this.conversionService = conversionService;
    this.propertyMappings = tableMapping.getPropertyMappings();
    this.mappedObjConstructor = tableMapping.getMappedObjConstructor();
    // offset is used for multi entity query resultSet
    this.startIndex = offset;
    this.endIndex = propertyMappings.length + offset - 1;
  }

  @Override
  public T mapRow(ResultSet rs, int rowNumber) throws SQLException {
    T obj = null;
    try {
      obj = mappedObjConstructor.newInstance();
      // since the sql columns were generated using the property mappings the
      // resultset columns will be in same order.
      for (int index = startIndex; index <= endIndex; index++) {
        // propertyMappings index starts at 0
        PropertyMapping propMapping = propertyMappings[index - startIndex];
        Object value = getResultSetValue(rs, index, propMapping);
        propMapping.getWriteMethod().invoke(obj, value);
      }
    } catch (Exception e) {
      throw new MapperException(e.getMessage(), e);
    }
    return obj;
  }

  /*
   * Same logic as Spring's JdbcUtil.getResultSetValue().
   * JdbcUtil.getResultSetValue() logic has been proven over the years, retaining
   * its logic but changed the structure to use 'switch' statement with enums
   * instead of the bunch of if/else's for performance reasons. As was the goal,
   * java compiled the switch statement into a 'tableswitch', which means the
   * program will jump directly to the correct 'case' block in one step.
   *
   * Also this method is eligible for jit 'inline (hot)' by making its size less
   * than 325 bytes
   */
  private Object getResultSetValue(ResultSet rs, int index, PropertyMapping propMapping)
      throws SQLException {
    Object value;
    switch (propMapping.getResultSetProcessingType()) {
      case ResultSetProcessingType.BOOLEAN:
        value = rs.getBoolean(index);
        break;
      case ResultSetProcessingType.BYTE:
        value = rs.getByte(index);
        break;
      case ResultSetProcessingType.SHORT:
        value = rs.getShort(index);
        break;
      case ResultSetProcessingType.INTEGER:
        value = rs.getInt(index);
        break;
      case ResultSetProcessingType.LONG:
        value = rs.getLong(index);
        break;
      case ResultSetProcessingType.FLOAT:
        value = rs.getFloat(index);
        break;
      case ResultSetProcessingType.DOUBLE:
        value = rs.getDouble(index);
        break;
      case ResultSetProcessingType.NUMBER: // same as double
        value = rs.getDouble(index);
        break;
      case ResultSetProcessingType.STRING:
        return rs.getString(index);
      case ResultSetProcessingType.BIGDECIMAL:
        return rs.getBigDecimal(index);
      case ResultSetProcessingType.DATE:
        return rs.getDate(index);
      case ResultSetProcessingType.TIME:
        return rs.getTime(index);
      case ResultSetProcessingType.TIMESTAMP:
        return rs.getTimestamp(index);
      case ResultSetProcessingType.UTILDATE: // java.util.Date. same as timestamp
        return rs.getTimestamp(index);
      case ResultSetProcessingType.BYTEARRAY:
        return rs.getBytes(index);
      case ResultSetProcessingType.ENUM:
        return getEnumValue(rs, index, propMapping);
      case ResultSetProcessingType.CUSTOMCONVERTER:
        return getCustomConvertedValue(rs, index, propMapping);
      default:
        return getUnknownTypeValue(rs, index, propMapping);
    }
    // Perform was-null check if necessary (for results that the JDBC driver returns
    // as primitives).
    return (rs.wasNull() ? null : value);
  }

  private Object getEnumValue(ResultSet rs, int index, PropertyMapping propMapping)
      throws SQLException {
    // Enums are represented as a String in simpleJdbcMapper.
    Object val = rs.getObject(index);
    if (val != null) {
      if (val instanceof String) {
        return conversionService.convert(val, propMapping.getPropertyType());
      } else {
        // for example, on Postgres: getObject returns a PGObject, but we need a String
        val = rs.getString(index);
        return conversionService.convert(val, propMapping.getPropertyType());
      }
    } else {
      return null;
    }
  }

  @SuppressWarnings("unchecked")
  private Object getCustomConvertedValue(ResultSet rs, int index, PropertyMapping propMapping)
      throws SQLException {
    Object val = rs.getObject(index);
    return val == null ? null : propMapping.getConverter().convertToEntityAttribute(val);
  }

  private Object getUnknownTypeValue(ResultSet rs, int index, PropertyMapping propMapping)
      throws SQLException {
    // Some unknown type desired -> rely on getObject.
    try {
      return rs.getObject(index, propMapping.getPropertyType());
    } catch (SQLFeatureNotSupportedException | AbstractMethodError ex) {
      if (logger.isDebugEnabled()) {
        logger.debug("JDBC driver does not support JDBC 4.1 'getObject(int, Class)' method", ex);
      }
    } catch (SQLException ex) {
      if (logger.isDebugEnabled()) {
        logger.debug(
            "JDBC driver has limited support for 'getObject(int, Class)' with column type: "
                + propMapping.getPropertyType().getName(),
            ex);
      }
    }
    // Corresponding SQL types for JSR-310, Will use conversionService to convert
    String typeName = propMapping.getPropertyType().getSimpleName();
    Object val = null;
    switch (typeName) {
      case "LocalDate":
        val = rs.getDate(index);
        break;
      case "LocalTime":
        val = rs.getTime(index);
        break;
      case "LocalDateTime":
        val = rs.getTimestamp(index);
        break;
      // Fall back to getObject without type specification.
      default:
        val = JdbcUtils.getResultSetValue(rs, index);
    }
    // Use conversionService to convert.
    return val == null ? null : conversionService.convert(val, propMapping.getPropertyType());
  }

  @Override
  public String toString() {
    return "%s startIndex: %d  endIndex: %d"
        .formatted(mappedObjConstructor.getName(), startIndex, endIndex);
  }
}
