package io.github.simplejdbcmapper.core;

import org.springframework.util.StringUtils;

/**
 * Creates the full select clause
 *
 * @author Antony Joseph
 */
class Select implements WithClauses {
  private String sqlColumns;

  public Select(String sqlColumns) {
    this.sqlColumns = sqlColumns;
  }

  public String withClauses(String clauses) {
    if (!StringUtils.hasText(clauses)) {
      throw new IllegalArgumentException("clauses is empty");
    }
    return "SELECT %s %s".formatted(sqlColumns, clauses);
  }
}
